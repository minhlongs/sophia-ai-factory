# Phase 01: Database Analytics Schema

## Context

**Related Docs:**
- Usage Metering Schema: `supabase/migrations/20260307-usage-metering-schema-updates.sql`
- Summary Tables: `supabase/migrations/20260307-create-usage-summary-tables.sql`
- RaaS Licenses: `docs/migrations/raas-licenses-schema.sql`

**Current State:**
- `usage_events` table tracks raw API usage
- `usage_hourly_summary` and `usage_daily_summary` for aggregated metrics
- `raas_licenses` table with Polar/Stripe customer linkage
- Missing: Revenue tracking, ROI metrics, cohort analysis tables

## Requirements

### Functional
1. Track revenue per license/tier
2. Store ROI metrics (projected vs actual)
3. Cohort analysis for user retention
4. Revenue reconciliation with Polar/Shopee

### Non-Functional
1. RLS policies for data isolation
2. Indexes for fast analytics queries
3. Partitioning for large datasets (if needed)

## Implementation Steps

### Step 1: Create Revenue Metrics Tables

```sql
-- Revenue tracking per license
CREATE TABLE analytics_revenue_metrics (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  license_nonce TEXT NOT NULL,
  tenant_id TEXT NOT NULL,

  -- Time period
  period_type TEXT NOT NULL CHECK (period_type IN ('hourly', 'daily', 'monthly')),
  period_start TIMESTAMPTZ NOT NULL,
  period_end TIMESTAMPTZ NOT NULL,

  -- Revenue data
  gross_revenue DECIMAL(12,2) DEFAULT 0,
  net_revenue DECIMAL(12,2) DEFAULT 0,
  refunds DECIMAL(12,2) DEFAULT 0,

  -- Usage data
  total_credits_used INTEGER DEFAULT 0,
  total_requests INTEGER DEFAULT 0,

  -- Tier at period
  tier TEXT NOT NULL,

  -- External linkage
  polar_order_id TEXT,
  stripe_subscription_id TEXT,

  -- Metadata
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- Unique index for idempotent inserts
CREATE UNIQUE INDEX idx_revenue_metrics_unique
  ON analytics_revenue_metrics(license_nonce, period_type, period_start);
```

### Step 2: Create ROI Metrics Table

```sql
-- ROI tracking per license
CREATE TABLE analytics_roi_metrics (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  license_nonce TEXT NOT NULL,
  tenant_id TEXT NOT NULL,

  -- Time period
  period_start TIMESTAMPTZ NOT NULL,
  period_end TIMESTAMPTZ NOT NULL,

  -- Investment (cost)
  subscription_cost DECIMAL(12,2) NOT NULL,
  api_costs DECIMAL(12,2) DEFAULT 0,
  total_investment DECIMAL(12,2) NOT NULL,

  -- Returns (revenue generated by user's campaigns)
  estimated_campaign_revenue DECIMAL(12,2) DEFAULT 0,
  video_views BIGINT DEFAULT 0,
  click_throughs BIGINT DEFAULT 0,
  conversions BIGINT DEFAULT 0,

  -- ROI calculation
  roi_percentage DECIMAL(8,2) DEFAULT 0,
  payback_period_days INTEGER,

  -- Metadata
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE UNIQUE INDEX idx_roi_metrics_unique
  ON analytics_roi_metrics(license_nonce, period_start);
```

### Step 3: Create User Cohorts Table

```sql
-- Cohort analysis for user retention
CREATE TABLE analytics_user_cohorts (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id TEXT NOT NULL,

  -- Cohort assignment
  cohort_month DATE NOT NULL, -- First purchase month
  cohort_tier TEXT NOT NULL,

  -- Period tracking
  period_number INTEGER NOT NULL, -- Months since cohort start
  period_start DATE NOT NULL,

  -- Activity status
  is_active BOOLEAN NOT NULL,
  video_count INTEGER DEFAULT 0,
  revenue_generated DECIMAL(12,2) DEFAULT 0,

  -- Metadata
  created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE UNIQUE INDEX idx_cohorts_unique
  ON analytics_user_cohorts(user_id, cohort_month, period_number);
```

### Step 4: Create Revenue Reconciliation Table

```sql
-- Reconciliation between Polar/Stripe and internal tracking
CREATE TABLE analytics_revenue_reconciliation (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),

  -- External source
  source TEXT NOT NULL CHECK (source IN ('polar', 'stripe')),
  external_order_id TEXT NOT NULL,
  external_customer_id TEXT NOT NULL,

  -- Internal linkage
  license_nonce TEXT,
  tenant_id TEXT,

  -- Amounts
  external_amount DECIMAL(12,2) NOT NULL,
  internal_amount DECIMAL(12,2) NOT NULL,
  difference DECIMAL(12,2) DEFAULT 0,

  -- Status
  status TEXT NOT NULL CHECK (status IN ('matched', 'discrepancy', 'unlinked')),
  reconciled_at TIMESTAMPTZ,
  reconciled_by TEXT,

  -- Metadata
  created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE INDEX idx_reconciliation_source
  ON analytics_revenue_reconciliation(source, external_order_id);
CREATE INDEX idx_reconciliation_status
  ON analytics_revenue_reconciliation(status);
```

### Step 5: Add RLS Policies

```sql
-- Enable RLS
ALTER TABLE analytics_revenue_metrics ENABLE ROW LEVEL SECURITY;
ALTER TABLE analytics_roi_metrics ENABLE ROW LEVEL SECURITY;
ALTER TABLE analytics_user_cohorts ENABLE ROW LEVEL SECURITY;
ALTER TABLE analytics_revenue_reconciliation ENABLE ROW LEVEL SECURITY;

-- Users can view their own data
CREATE POLICY "Users can view own revenue metrics"
  ON analytics_revenue_metrics
  FOR SELECT
  USING (tenant_id = auth.uid()::TEXT);

CREATE POLICY "Users can view own ROI metrics"
  ON analytics_roi_metrics
  FOR SELECT
  USING (tenant_id = auth.uid()::TEXT);

CREATE POLICY "Users can view own cohort data"
  ON analytics_user_cohorts
  FOR SELECT
  USING (user_id = auth.uid()::TEXT);

-- Service role full access
CREATE POLICY "Service role full access revenue"
  ON analytics_revenue_metrics FOR ALL
  USING (auth.jwt()->>'role' = 'service_role');

CREATE POLICY "Service role full access ROI"
  ON analytics_roi_metrics FOR ALL
  USING (auth.jwt()->>'role' = 'service_role');

CREATE POLICY "Service role full access cohorts"
  ON analytics_user_cohorts FOR ALL
  USING (auth.jwt()->>'role' = 'service_role');

-- Admin role can view all (for analytics dashboard)
CREATE POLICY "Admin can view all revenue"
  ON analytics_revenue_metrics
  FOR SELECT
  USING (auth.jwt()->>'role' = 'admin');
```

### Step 6: Create Indexes for Performance

```sql
-- Revenue metrics indexes
CREATE INDEX idx_revenue_tenant_period
  ON analytics_revenue_metrics(tenant_id, period_start DESC);
CREATE INDEX idx_revenue_license
  ON analytics_revenue_metrics(license_nonce);
CREATE INDEX idx_revenue_tier
  ON analytics_revenue_metrics(tier, period_start DESC);

-- ROI metrics indexes
CREATE INDEX idx_roi_tenant
  ON analytics_roi_metrics(tenant_id, period_start DESC);
CREATE INDEX idx_roi_license
  ON analytics_roi_metrics(license_nonce);

-- Cohort indexes
CREATE INDEX idx_cohorts_month
  ON analytics_user_cohorts(cohort_month);
CREATE INDEX idx_cohorts_tier
  ON analytics_user_cohorts(cohort_tier, cohort_month);
```
